cmake_minimum_required(VERSION 3.22)
project(GUMMY VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GUMMY_BUILD_PLUGIN "Build native JUCE plugin and integration tests" ON)
option(GUMMY_BUILD_TESTS "Build automated tests" ON)
set(GUMMY_JUCE_PATH "" CACHE PATH "Existing JUCE 8.0.6 checkout (otherwise fetched)")
if(GUMMY_BUILD_PLUGIN)
 if(GUMMY_JUCE_PATH)
  add_subdirectory("${GUMMY_JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
 else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
 endif()
 set(formats VST3 Standalone)
 if(APPLE)
  list(PREPEND formats AU)
 endif()
 juce_add_plugin(GUMMY COMPANY_NAME "GUMMY Audio" BUNDLE_ID "com.gummyaudio.gummy"
  PLUGIN_MANUFACTURER_CODE GmAu PLUGIN_CODE Gmmy PRODUCT_NAME GUMMY
  FORMATS ${formats} IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
  VST3_CATEGORIES Fx Dynamics Distortion Modulation EQ AU_MAIN_TYPE kAudioUnitType_Effect)
 juce_generate_juce_header(GUMMY)
 target_sources(GUMMY PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/Parameters/Parameters.cpp Source/Presets/FactoryPresets.cpp)
 target_include_directories(GUMMY PUBLIC Source)
 target_compile_definitions(GUMMY PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(GUMMY PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
endif()
if(GUMMY_BUILD_TESTS)
 enable_testing()
 add_executable(GummyDSPTests Tests/DSPTests.cpp Tests/DynamicsTests.cpp Tests/AdvancedTests.cpp)
 target_include_directories(GummyDSPTests PRIVATE Source)
 add_test(NAME DSP COMMAND GummyDSPTests)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME MacScriptContracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_mac_scripts.py")
 endif()
 if(GUMMY_BUILD_PLUGIN)
  juce_add_console_app(GummyPluginTests PRODUCT_NAME "GUMMY Integration Tests")
  target_sources(GummyPluginTests PRIVATE Tests/PluginTests.cpp)
  target_link_libraries(GummyPluginTests PRIVATE GUMMY)
  target_include_directories(GummyPluginTests PRIVATE $<TARGET_PROPERTY:GUMMY,INCLUDE_DIRECTORIES>)
  target_compile_definitions(GummyPluginTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  add_test(NAME PluginIntegration COMMAND GummyPluginTests)
  juce_add_console_app(GummyFormatTests PRODUCT_NAME "GUMMY VST3 Host Tests")
  target_sources(GummyFormatTests PRIVATE Tests/PluginFormatTests.cpp)
  target_compile_definitions(GummyFormatTests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(GummyFormatTests PRIVATE juce::juce_audio_processors juce::juce_gui_extra)
  add_dependencies(GummyFormatTests GUMMY_VST3)
  add_test(NAME VST3BinaryIntegration COMMAND GummyFormatTests "$<TARGET_FILE_DIR:GUMMY_VST3>/../..")
 endif()
endif()
